## API文档

### REST API
所有API端点前缀：`/api/v1`

#### 任务管理
- `POST /tasks` - 创建新任务
- `GET /tasks` - 获取任务列表
- `GET /tasks/{id}` - 获取任务详情
- `DELETE /tasks/{id}` - 取消任务
- `GET /tasks/{id}/files/{filename}` - 下载任务文件

#### 节点管理
- `GET /nodes` - 获取节点列表
- `GET /nodes/{id}` - 获取节点详情
- `POST /nodes/{id}/restart` - 重启节点

#### 系统管理
- `GET /system/status` - 获取系统状态
- `GET /system/metrics` - 获取系统指标
- `GET /system/logs` - 获取系统日志

### WebSocket API
- 连接地址：`ws://{host}:{port}/ws`
- 支持消息类型：任务状态、节点状态、系统通知

## 7. 初始化脚本

创建一个初始化脚本 `scripts/init.sh`：

```bash
#!/bin/bash
# RAMGEO分布式系统初始化脚本

set -e

echo "🚀 RAMGEO分布式系统初始化开始..."

# 检查依赖
check_dependencies() {
    echo "📦 检查系统依赖..."
    
    # 检查Python
    if ! command -v python3 &> /dev/null; then
        echo "❌ Python3未安装"
        exit 1
    fi
    
    # 检查pip
    if ! command -v pip3 &> /dev/null; then
        echo "❌ pip3未安装"
        exit 1
    fi
    
    # 检查Node.js (可选)
    if ! command -v node &> /dev/null; then
        echo "⚠️  Node.js未安装（Web界面需要）"
    fi
    
    echo "✅ 依赖检查通过"
}

# 安装Python依赖
install_python_deps() {
    echo "📦 安装Python依赖..."
    
    pip3 install --upgrade pip
    pip3 install -r requirements.txt
    
    echo "✅ Python依赖安装完成"
}

# 安装Node.js依赖
install_node_deps() {
    if [ -d "web_client" ] && [ -f "web_client/package.json" ]; then
        echo "📦 安装Node.js依赖..."
        
        cd web_client
        npm install
        cd ..
        
        echo "✅ Node.js依赖安装完成"
    fi
}

# 初始化数据库
init_database() {
    echo "🗄️  初始化数据库..."
    
    # 检查PostgreSQL是否可用
    if command -v psql &> /dev/null; then
        if [ -f "migrations/001_initial.sql" ]; then
            echo "📊 创建数据库表..."
            psql -U postgres -f migrations/001_initial.sql
        fi
    else
        echo "⚠️  PostgreSQL未安装，跳过数据库初始化"
    fi
    
    echo "✅ 数据库初始化完成"
}

# 创建配置文件
create_configs() {
    echo "⚙️  创建配置文件..."
    
    # 创建示例配置文件
    if [ ! -f "config/proxy.yaml" ]; then
        cp "config/proxy.example.yaml" "config/proxy.yaml"
        echo "📝 创建proxy.yaml配置文件"
    fi
    
    if [ ! -f "config/node.yaml" ]; then
        cp "config/node.example.yaml" "config/node.yaml"
        echo "📝 创建node.yaml配置文件"
    fi
    
    if [ ! -f ".env" ]; then
        cp ".env.example" ".env"
        echo "📝 创建.env环境变量文件"
        echo "⚠️  请编辑.env文件，设置正确的配置值"
    fi
    
    echo "✅ 配置文件创建完成"
}

# 创建目录结构
create_directories() {
    echo "📁 创建目录结构..."
    
    mkdir -p logs/proxy
    mkdir -p logs/node
    mkdir -p uploads
    mkdir -p temp
    
    echo "✅ 目录结构创建完成"
}

# 主函数
main() {
    echo "=========================================="
    echo "    RAMGEO分布式系统初始化"
    echo "=========================================="
    
    check_dependencies
    install_python_deps
    install_node_deps
    create_configs
    create_directories
    init_database
    
    echo ""
    echo "=========================================="
    echo "✅ 初始化完成！"
    echo ""
    echo "接下来您可以："
    echo "1. 编辑配置文件：config/*.yaml 和 .env"
    echo "2. 启动代理服务器：python proxy_server/main.py"
    echo "3. 启动计算节点：python node_server/main.py"
    echo "4. 启动Web界面：cd web_client && npm run dev"
    echo ""
    echo "详细文档请参考 docs/ 目录"
    echo "=========================================="
}

# 运行主函数
main
```

```yaml
name: Deploy to GitHub Pages

on:
  push:
    branches: [ main ]

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Setup Node.js
      uses: actions/setup-node@v3
      with:
        node-version: '16'
    
    - name: Install dependencies
      run: |
        cd web_client
        npm ci
    
    - name: Build
      run: |
        cd web_client
        npm run build
    
    - name: Deploy to GitHub Pages
      uses: peaceiris/actions-gh-pages@v3
      with:
        github_token: ${{ secrets.GITHUB_TOKEN }}
        publish_dir: ./web_client/dist
```
